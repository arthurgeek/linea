fn make(self: Cell) -> Cell {
    let value = self.value
    let named = cells.Cell[UInt8] { value }
    let nested = ArrayBox[[UInt8; 3]] { value = [1, 2, 3] }
    if named.value == nested.value[0] { consume(named) }
    match self { Cell { value } => Self { value } }
}
