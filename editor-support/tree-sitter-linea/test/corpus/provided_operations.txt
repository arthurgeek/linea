==================

trait Gauge {
    fn size(self: Self) -> UInt16

    fn twice(self: Self) -> UInt16 {
        self.size() * 2
    }

    test "twice doubles a reading" {
        check()
    }

    test trap "twice past the largest reading traps" {
        check()
    }
}
