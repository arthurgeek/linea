==================
A walk and a match arm take a tuple apart
==================

fn total(pairs: Vec[(UInt8, UInt8)]) -> UInt8 {
    var sum = 0
    for (a, b) in pairs {
        sum = sum + a + b
    }
    for read (a, _) in pairs {
        sum = sum + a
    }
    match (sum, 1) {
        (a, b) => a + b,
    }
}
