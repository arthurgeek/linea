fn compare(read left: Self, read right: Self) -> Bool {
    let ordinal = reflect.ordinal_of(left)
    match read left {
        reflect.case => match read right {
            reflect.same_case => reflect.case.ordinal == ordinal,
            reflect.other_case => false,
        },
    }
}
